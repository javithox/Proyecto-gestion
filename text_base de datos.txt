
-- Base de datos: Catastro de Equipos Industriales
-- PostgreSQL 15+
-- Este archivo crea tablas dentro de la base de datos
-- a la que se conecta PostgreSQL; no crea la base de datos en sí.

CREATE TABLE IF NOT EXISTS ubicaciones (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre VARCHAR(120) NOT NULL,
    codigo VARCHAR(40) NOT NULL UNIQUE,
    tipo VARCHAR(30) NOT NULL
        CHECK (tipo IN ('PLANTA', 'AREA', 'LINEA', 'SECTOR', 'BODEGA')),
    descripcion TEXT,
    ubicacion_padre_id BIGINT REFERENCES ubicaciones(id)
        ON DELETE RESTRICT,
    activo BOOLEAN NOT NULL DEFAULT TRUE,
    creado_en TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS equipos (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    codigo VARCHAR(50) NOT NULL UNIQUE,
    nombre VARCHAR(150) NOT NULL,
    descripcion TEXT,
    tipo VARCHAR(80) NOT NULL,
    marca VARCHAR(100),
    modelo VARCHAR(100),
    numero_serie VARCHAR(100) UNIQUE,
    fabricante VARCHAR(150),
    anio_fabricacion SMALLINT
        CHECK (anio_fabricacion IS NULL OR anio_fabricacion BETWEEN 1800 AND 2200),
    fecha_adquisicion DATE,
    fecha_instalacion DATE,
    potencia_kw NUMERIC(12,3)
        CHECK (potencia_kw IS NULL OR potencia_kw >= 0),
    voltaje NUMERIC(12,2)
        CHECK (voltaje IS NULL OR voltaje >= 0),
    criticidad VARCHAR(20) NOT NULL DEFAULT 'MEDIA'
        CHECK (criticidad IN ('BAJA', 'MEDIA', 'ALTA', 'CRITICA')),
    estado VARCHAR(25) NOT NULL DEFAULT 'OPERATIVO'
        CHECK (estado IN (
            'OPERATIVO', 'EN_MANTENCION', 'FUERA_SERVICIO',
            'EN_REPARACION', 'DADO_DE_BAJA'
        )),
    ubicacion_id BIGINT REFERENCES ubicaciones(id)
        ON DELETE RESTRICT,
    observaciones TEXT,
    creado_en TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS componentes (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    equipo_id BIGINT NOT NULL REFERENCES equipos(id)
        ON DELETE CASCADE,
    codigo VARCHAR(50) NOT NULL,
    nombre VARCHAR(150) NOT NULL,
    tipo VARCHAR(80),
    marca VARCHAR(100),
    modelo VARCHAR(100),
    numero_serie VARCHAR(100),
    fecha_instalacion DATE,
    vida_util_meses INTEGER
        CHECK (vida_util_meses IS NULL OR vida_util_meses > 0),
    estado VARCHAR(25) NOT NULL DEFAULT 'OPERATIVO'
        CHECK (estado IN (
            'OPERATIVO', 'EN_MANTENCION', 'FUERA_SERVICIO', 'REEMPLAZADO'
        )),
    observaciones TEXT,
    creado_en TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT uq_componente_codigo_equipo UNIQUE (equipo_id, codigo)
);

CREATE TABLE IF NOT EXISTS historial_equipos (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    equipo_id BIGINT NOT NULL REFERENCES equipos(id)
        ON DELETE RESTRICT,
    accion VARCHAR(20) NOT NULL
        CHECK (accion IN ('CREACION', 'ACTUALIZACION', 'ELIMINACION')),
    fecha TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    usuario VARCHAR(150),
    detalle TEXT,
    valores_anteriores JSONB,
    valores_nuevos JSONB
);

CREATE INDEX IF NOT EXISTS idx_equipos_ubicacion
    ON equipos(ubicacion_id);

CREATE INDEX IF NOT EXISTS idx_equipos_estado
    ON equipos(estado);

CREATE INDEX IF NOT EXISTS idx_equipos_criticidad
    ON equipos(criticidad);

CREATE INDEX IF NOT EXISTS idx_componentes_equipo
    ON componentes(equipo_id);

CREATE INDEX IF NOT EXISTS idx_historial_equipo_fecha
    ON historial_equipos(equipo_id, fecha DESC);

-- Actualiza automáticamente la fecha de modificación.
CREATE OR REPLACE FUNCTION actualizar_fecha_modificacion()
RETURNS TRIGGER AS $$
BEGIN
    NEW.actualizado_en := CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_ubicaciones_actualizado
    ON ubicaciones;

CREATE TRIGGER trg_ubicaciones_actualizado
BEFORE UPDATE ON ubicaciones
FOR EACH ROW EXECUTE FUNCTION actualizar_fecha_modificacion();

DROP TRIGGER IF EXISTS trg_equipos_actualizado
    ON equipos;

CREATE TRIGGER trg_equipos_actualizado
BEFORE UPDATE ON equipos
FOR EACH ROW EXECUTE FUNCTION actualizar_fecha_modificacion();

DROP TRIGGER IF EXISTS trg_componentes_actualizado
    ON componentes;

CREATE TRIGGER trg_componentes_actualizado
BEFORE UPDATE ON componentes
FOR EACH ROW EXECUTE FUNCTION actualizar_fecha_modificacion();

-- Registra automáticamente la creación, modificación y eliminación
-- de la ficha del equipo en historial_equipos.
CREATE OR REPLACE FUNCTION registrar_historial_equipo()
RETURNS TRIGGER AS $$
DECLARE
    v_equipo_id BIGINT;
    v_accion VARCHAR(20);
    v_antes JSONB;
    v_despues JSONB;
BEGIN
    IF TG_OP = 'INSERT' THEN
        v_equipo_id := NEW.id;
        v_accion := 'CREACION';
        v_antes := NULL;
        v_despues := to_jsonb(NEW);
    ELSIF TG_OP = 'UPDATE' THEN
        v_equipo_id := NEW.id;
        v_accion := 'ACTUALIZACION';
        v_antes := to_jsonb(OLD);
        v_despues := to_jsonb(NEW);
    ELSE
        v_equipo_id := OLD.id;
        v_accion := 'ELIMINACION';
        v_antes := to_jsonb(OLD);
        v_despues := NULL;
    END IF;

    INSERT INTO historial_equipos (
        equipo_id, accion, detalle, valores_anteriores, valores_nuevos
    )
    VALUES (
        v_equipo_id,
        v_accion,
        'Cambio automático de ficha de equipo',
        v_antes,
        v_despues
    );

    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_historial_equipos
    ON equipos;

CREATE TRIGGER trg_historial_equipos
AFTER INSERT OR UPDATE OR DELETE ON equipos
FOR EACH ROW EXECUTE FUNCTION registrar_historial_equipo();
